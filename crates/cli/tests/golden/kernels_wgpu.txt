== rmsnorm (bf0b2f2e6d391de1)
target Wgsl cc=0.0 sms=0 warp=32 threads=256 smem=16384 graphs=false pdl=false features=[Subgroups]
kernel rmsnorm threads=256 smem=1024
  param x F32 Activation Read
  param weight F32 Weight Read
  param out F32 Activation Write
  rmsnorm width=1024 eps=1e-6 sum=Subgroup

@group(0) @binding(0) var<storage, read> x: array<f32>;
@group(0) @binding(1) var<storage, read> weight: array<f32>;
@group(0) @binding(2) var<storage, read_write> out: array<f32>;
struct P { a: vec4<u32>, b: vec4<u32> };
@group(0) @binding(3) var<uniform> p: P;

const WIDTH: u32 = 1024u;
const EPS: f32 = 1e-6;
var<workgroup> red: array<f32, 128>;

@compute @workgroup_size(256)
fn main(@builtin(workgroup_id) wid: vec3<u32>, @builtin(local_invocation_index) li: u32, @builtin(subgroup_id) sg: u32, @builtin(subgroup_invocation_id) lane: u32, @builtin(num_subgroups) sgs: u32) {
    let t = wid.x;
    let src = (p.a.x + t) * WIDTH;
    let dst = (p.a.y + t) * WIDTH;
    var ss = 0.0;
    for (var i = li; i < WIDTH; i += 256u) {
        let v = x[src + i];
        ss += v * v;
    }
    let part = subgroupAdd(ss);
    if (lane == 0u) {
        red[sg] = part;
    }
    workgroupBarrier();
    var total = 0.0;
    for (var i = 0u; i < sgs; i++) {
        total += red[i];
    }
    let inv = inverseSqrt(total / f32(WIDTH) + EPS);
    for (var i = li; i < WIDTH; i += 256u) {
        out[dst + i] = weight[i] * (x[src + i] * inv);
    }
}

== swiglu (c99094cd65d6126a)
target Wgsl cc=0.0 sms=0 warp=32 threads=256 smem=16384 graphs=false pdl=false features=[Subgroups]
kernel swiglu threads=256 smem=0
  param gate_up F32 Activation Read
  param out F32 Activation Write
  gated_activation width=3072 act=Silu

@group(0) @binding(0) var<storage, read> gate_up: array<f32>;
@group(0) @binding(1) var<storage, read_write> out: array<f32>;
struct P { a: vec4<u32>, b: vec4<u32> };
@group(0) @binding(2) var<uniform> p: P;

const WIDTH: u32 = 3072u;

@compute @workgroup_size(256)
fn main(@builtin(global_invocation_id) gid: vec3<u32>) {
    let idx = gid.x + gid.y * 8388608u;
    if (idx >= p.a.x * WIDTH) {
        return;
    }
    let t = idx / WIDTH;
    let i = idx % WIDTH;
    let g = gate_up[t * 2u * WIDTH + i * 2u];
    let a = g / (1.0 + exp(-g));
    out[idx] = a * gate_up[t * 2u * WIDTH + i * 2u + 1u];
}
